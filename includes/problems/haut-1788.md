??? problem "展开：题意、思路、证明、复杂度与 C++14 代码"
    **题目**：万分之一的光(hard version)

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1788>){ .problem-source target="_blank" rel="noopener" }

    **题意**：根据给定输入，一个整数，亮着的路灯的数量。

    **核心思路**：万分之一的光(hard version) 题意： 同上 解法： 相比于easy version增加了数据范围。对楼房先按横坐标排序，然后判断排序后的楼之间是否相邻（优先按照纵坐标排序，再按照横坐标排序，这样对比数组中相邻的楼房是否相邻即可），然后按照纵坐标排序，再判断排序后的楼之间是否相邻（同理），由于每条相邻的边上都有4栈路灯，所以只要将答案乘4即可。（使用快速排序，时间复杂度（n*logn））

    **正确性**：扫描前保存的信息与已经处理的输入完全对应。；每读入一个元素，分支覆盖其全部情况并正确更新累计状态。；扫描结束时所有输入恰好处理一次，累计状态即为所求。

    **复杂度**：通常为 O(n log n)，主要来自排序

    **易错点**：比较器必须满足严格弱序；相等时处理题目指定的次关键字。

    - 核验状态：C++14 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++14 实现**

    ```cpp linenums="1"
    #include <bits/stdc++.h>
    using namespace std;
    const int MAXN = 1e5 + 5;
    struct lou {
        int x, y;
    } a[MAXN];
    int n, k, ans;
    bool cmp1(lou a, lou b) {
        if (a.x != b.x)
            return a.x < b.x;
        else
            return a.y < b.y;
    }
    bool cmp2(lou a, lou b) {
        if (a.y != b.y)
            return a.y < b.y;
        else
            return a.x < b.x;
    }
    int main() {
        scanf("%d%d", &n, &k);
        for (int i = 1; i <= k; i++) {
            int x, y;
            scanf("%d%d", &x, &y);
            a[i].x = x;
            a[i].y = y;
        }
        sort(a + 1, a + k + 1, cmp1);
        for (int i = 1; i < k; i++)
            if (a[i].x == a[i + 1].x && a[i].y == a[i + 1].y - 1)
                ans++;
        sort(a + 1, a + k + 1, cmp2);
        for (int i = 1; i < k; i++)
            if (a[i].y == a[i + 1].y && a[i].x == a[i + 1].x - 1)
                ans++;
        printf("%d", ans * 4);
        return 0; // qwq
    }
    ```
