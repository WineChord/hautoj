??? problem "展开：题意、思路、证明、复杂度与 C++17 代码"
    **题目**：zp与火系道馆（对小19的再次关怀）

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1511>){ .problem-source target="_blank" rel="noopener" }

    **题意**：现在给你经过若干次1，2变换后的这n-3个四元组，请还原这个序列.如果有多个答案，请输出字典序最小的答案 。

    **核心思路**：原序列端点只出现在一个四元组中，次端点出现两次，第三个位置 出现三次，内部位置出现四次。选择两个频次为 1 的数中较小者作为 p1，可在正序与逆序中取得字典序更小者；其唯一四元组内其余三个数 按全局频次 2、3、4 排成 p2,p3,p4。之后每次寻找尚未使用且包含 最近三个数的四元组，第四个数就是下一项。

    **正确性**：构造只使用题目允许的元素与操作。；每一步都维持长度、取值、相邻关系或总和等约束。；结束时规模达到要求，因此输出是完整合法答案。

    **复杂度**：每组 O(n²) 时间、O(n) 空间

    **易错点**：样例只是一个合法答案；本地检查约束，不要逐字比对。

    - 核验状态：C++17 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++17 实现**

    ```cpp linenums="1"
    #include <bits/stdc++.h>
    using namespace std;
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int T;
        cin >> T;
        while (T--) {
            int n;
            cin >> n;
            int groups = n - 3;
            vector<array<int, 4>> tuple(groups);
            vector<int> freq(n + 1, 0);
            for (auto& group : tuple) {
                for (int& x : group) {
                    cin >> x;
                    ++freq[x];
                }
            }
            int first = n + 1;
            for (int value = 1; value <= n; ++value) {
                if (freq[value] == 1)
                    first = min(first, value);
            }
            int initial = -1;
            for (int i = 0; i < groups; ++i) {
                if (find(tuple[i].begin(), tuple[i].end(), first) != tuple[i].end()) {
                    initial = i;
                    break;
                }
            }
            vector<int> ans = {first};
            vector<int> rest;
            for (int x : tuple[initial])
                if (x != first)
                    rest.push_back(x);
            sort(rest.begin(), rest.end(), [&](int a, int b) {
                if (freq[a] != freq[b])
                    return freq[a] < freq[b];
                return a < b;
            });
            ans.insert(ans.end(), rest.begin(), rest.end());
            vector<bool> used(groups, false);
            used[initial] = true;
            while ((int)ans.size() < n) {
                array<int, 3> last = {ans[ans.size() - 3], ans[ans.size() - 2],
                                      ans[ans.size() - 1]};
                for (int i = 0; i < groups; ++i) {
                    if (used[i])
                        continue;
                    bool containsAll = true;
                    for (int x : last) {
                        if (find(tuple[i].begin(), tuple[i].end(), x) == tuple[i].end()) {
                            containsAll = false;
                        }
                    }
                    if (!containsAll)
                        continue;
                    for (int x : tuple[i]) {
                        if (x != last[0] && x != last[1] && x != last[2]) {
                            ans.push_back(x);
                            break;
                        }
                    }
                    used[i] = true;
                    break;
                }
            }
            for (int i = 0; i < n; ++i) {
                if (i)
                    cout << ' ';
                cout << ans[i];
            }
            cout << '\n';
        }
        return 0;
    }
    ```
