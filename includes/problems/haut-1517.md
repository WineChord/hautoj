    **C++14 实现**

    ```cpp linenums="1"
    #include <iostream>
    #include <algorithm>
    #define maxn 1005
    #define _for(i, a) for (LL i = 0; i < (a); ++i)
    using namespace std;
    typedef long long LL;
    LL n, k;
    LL P[maxn];
    LL p, q; // 最接近mid的分子和分母
    LL find(double mid) {
        LL cnt = 0;
        _for(i, n) {
            LL num = mid * P[i];                                  // 向下取整
            cnt += P[i] - num - 1;                                // 比mid小的分数
            if (p == -1 && q == -1 || p * P[i] > (num + 1) * q) { // 保存分子和分母
                p = num + 1;
                q = P[i];
            }
        }
        return cnt;
    }
    void sol() {
        cin >> n >> k;
        _for(i, n) cin >> P[i];
        double l = 0, r = 1;
        while (1) {
            p = -1, q = -1;
            double mid = (l + r) / 2;
            LL cnt = find(mid);
            if (cnt == k)
                break;
            if (cnt < k)
                r = mid;
            else
                l = mid;
        }
        cout << p << "/" << q << "\n";
    }
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        // freopen("in.txt", "r", stdin);
        sol();
        return 0;
    }
    ```
