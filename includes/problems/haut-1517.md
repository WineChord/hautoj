    **题目**：第k大分数（加强版）

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1517>){ .problem-source target="_blank" rel="noopener" }

    **题意**：请帮助小EF求出其中第 k 大的分数

    **核心思路**：每个质数 p\_i 产生 1/p\_i 到 (p\_i-1)/p\_i。二分实数阈值 x，分母为 p\_i 时，大于 x 的分数个数为 p\_i-1-floor(x×p\_i)，累加即可判断第 k 大分数位于阈值哪一侧。判定时同步保存刚好大于阈值的最小分数，分数大小使用交叉乘法比较，避免只靠浮点排序。

    **正确性**：推导把原过程转换为等价的公式或不变量。；算法直接计算该等价量，并使用足够宽的数据类型处理边界。；因此计算结果就是题目定义的答案。

    **复杂度**：O(n log(1/ε)) 时间、O(n) 空间

    **易错点**：确认循环边界，以及最终返回第一个可行还是最后一个不可行。；检查 gcd/lcm 的 0 值、负数和乘法溢出。；按上界估算中间量，相关变量不要退回 int。

    - 核验状态：C++14 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++14 实现**

    ```cpp linenums="1"
    #include <iostream>
    #include <algorithm>
    #define maxn 1005
    #define _for(i, a) for (LL i = 0; i < (a); ++i)
    using namespace std;
    typedef long long LL;
    LL n, k;
    LL P[maxn];
    LL p, q; // 最接近mid的分子和分母
    LL find(double mid) {
        LL cnt = 0;
        _for(i, n) {
            LL num = mid * P[i];                                  // 向下取整
            cnt += P[i] - num - 1;                                // 比mid小的分数
            if (p == -1 && q == -1 || p * P[i] > (num + 1) * q) { // 保存分子和分母
                p = num + 1;
                q = P[i];
            }
        }
        return cnt;
    }
    void sol() {
        cin >> n >> k;
        _for(i, n) cin >> P[i];
        double l = 0, r = 1;
        while (1) {
            p = -1, q = -1;
            double mid = (l + r) / 2;
            LL cnt = find(mid);
            if (cnt == k)
                break;
            if (cnt < k)
                r = mid;
            else
                l = mid;
        }
        cout << p << "/" << q << "\n";
    }
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(0);
        cout.tie(0);
        // freopen("in.txt", "r", stdin);
        sol();
        return 0;
    }
    ```
