??? problem "展开：题意、思路、证明、复杂度与 C++17 代码"
    **题目**：编程课

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1987>){ .problem-source target="_blank" rel="noopener" }

    **题意**：任务是模拟以上过程，确定做题的配对及顺序。

    **核心思路**：用 left/right 数组把当前仍在队列中的人维护成双向链表。优先队列 保存所有相邻且性别不同的候选对，按能力差、左端编号排序。弹出时若 两人已不相邻就丢弃；否则记录并删除二人，再把新形成的边界候选入堆。 这是“堆延迟删除＋数组链表”的典型组合。

    **正确性**：初始变量与题面初始状态相同。；若某一步前状态正确，本步按相同顺序和规则更新后仍保持一致。；由归纳法，全部步骤结束时程序状态与答案都正确。

    **复杂度**：O(n log n) 时间、O(n) 空间

    **易错点**：按规定顺序更新，避免本轮新状态在同一轮被重复使用。

    - 核验状态：C++17 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++17 实现**

    ```cpp linenums="1"
    #include <bits/stdc++.h>
    using namespace std;
    struct Candidate {
        int diff, left, right;
        bool operator>(const Candidate& other) const {
            return tie(diff, left, right) > tie(other.diff, other.left, other.right);
        }
    };
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n;
        cin >> n;
        string gender;
        cin >> gender;
        gender = " " + gender;
        vector<int> skill(n + 1), left(n + 1), right(n + 1);
        for (int i = 1; i <= n; ++i) {
            cin >> skill[i];
            left[i] = i - 1;
            right[i] = i + 1;
        }
        priority_queue<Candidate, vector<Candidate>, greater<Candidate>> heap;
        auto addCandidate = [&](int a, int b) {
            if (a >= 1 && b <= n && gender[a] != gender[b]) {
                heap.push({abs(skill[a] - skill[b]), a, b});
            }
        };
        for (int i = 1; i < n; ++i)
            addCandidate(i, i + 1);
        vector<pair<int, int>> ans;
        while (!heap.empty()) {
            Candidate cur = heap.top();
            heap.pop();
            int x = cur.left, y = cur.right;
            if (right[x] != y || left[y] != x)
                continue;
            ans.push_back({x, y});
            int nl = left[x], nr = right[y];
            if (nl >= 1)
                right[nl] = nr;
            if (nr <= n)
                left[nr] = nl;
            right[x] = left[y] = -1;
            addCandidate(nl, nr);
        }
        cout << ans.size() << '\n';
        for (auto [l, r] : ans) {
            cout << l << ' ' << r << '\n';
        }
        return 0;
    }
    ```
