    **题目**：资源箱

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1436>){ .problem-source target="_blank" rel="noopener" }

    **题意**：起重机所耗能的最大和最小的值及其取得最大和最小值的每一步操作(数据保证：只有唯一解)

    **核心思路**：把已经被勾过、目前位于某辆车顶部的箱子看成一个“整体块”。 再勾这辆车的下一个未勾箱时，整体块会和新箱一起搬走；连续从同一辆车 搬则只新增一个箱子，换边搬才会把已经积累的整体块再次计入耗能。 最小耗能一定是把总重量较小的那一堆逐箱搬到另一边：连续同向操作时， 每个箱子恰好被搬一次，总耗能就是该堆总重量。 最大耗能时应让两边尽量交替，使已勾箱整体被反复搬运。最终留在 B， 操作次数分别为 A:n、B:m-1；最终留在 A，则为 A:n-1、B:m。 对每种结局，多出来的同向操作必须放在开头，随后严格交替。模拟两种 唯一候选方案并取耗能较大者即可。总耗能可能超过 64 位，使用 __int128。

    **正确性**：每一步选择不占用额外未来资源，并达到当前可证明的最优边界。；任意最优解若首个选择不同，都可交换为本算法的选择而不使答案变差。；反复交换即可得到与算法相同的最优解，所以算法全局最优。

    **复杂度**：每组 O(n+m) 时间、O(n+m) 空间

    **易错点**：先尝试找反例；无法给出交换论证或下界时，不要把直觉当证明。；按规定顺序更新，避免本轮新状态在同一轮被重复使用。；样例只是一个合法答案；本地检查约束，不要逐字比对。；按上界估算中间量，相关变量不要退回 int。

    - 核验状态：C++17 编译通过；1 个可机读公开样例/合法构造校验通过

    ```cpp linenums="1"
    #include <bits/stdc++.h>
    using namespace std;
    using i128 = __int128_t;
    string toString(i128 x) {
        if (x == 0)
            return "0";
        string s;
        while (x > 0) {
            s.push_back(char('0' + x % 10));
            x /= 10;
        }
        reverse(s.begin(), s.end());
        return s;
    }
    struct Plan {
        i128 cost = 0;
        vector<pair<char, i128>> ops;
    };
    string alt(int countA, int countB, char last) {
        string labels;
        labels.reserve(countA + countB);
        if (last == 'A') {
            if (countA >= countB) {
                labels.append(countA - countB, 'A');
                for (int i = 0; i < countB; ++i)
                    labels += "BA";
            } else {
                labels.append(countB - countA, 'B');
                for (int i = 0; i < countA; ++i)
                    labels += "BA";
            }
        } else {
            if (countB >= countA) {
                labels.append(countB - countA, 'B');
                for (int i = 0; i < countA; ++i)
                    labels += "AB";
            } else {
                labels.append(countA - countB, 'A');
                for (int i = 0; i < countB; ++i)
                    labels += "AB";
            }
        }
        return labels;
    }
    Plan simulate(const vector<long long>& a, const vector<long long>& b,
                  const string& labels) {
        int ia = (int)a.size() - 1;
        int ib = (int)b.size() - 1;
        i128 hookedA = 0, hookedB = 0;
        Plan plan;
        plan.ops.reserve(labels.size());
        for (char side : labels) {
            if (side == 'A') {
                i128 moved = hookedA + a[ia--];
                hookedA = 0;
                hookedB += moved;
                plan.cost += moved;
                plan.ops.push_back({'A', moved});
            } else {
                i128 moved = hookedB + b[ib--];
                hookedB = 0;
                hookedA += moved;
                plan.cost += moved;
                plan.ops.push_back({'B', moved});
            }
        }
        return plan;
    }
    void printPlan(const Plan& plan) {
        cout << toString(plan.cost) << '\n';
        for (auto [side, weight] : plan.ops) {
            if (side == 'A')
                cout << "A>" << toString(weight) << ">B\n";
            else
                cout << "B>" << toString(weight) << ">A\n";
        }
    }
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int T;
        cin >> T;
        while (T--) {
            int n, m;
            cin >> n >> m;
            vector<long long> a(n), b(m);
            for (long long& x : a)
                cin >> x;
            for (long long& x : b)
                cin >> x;
            Plan endAtB = simulate(a, b, alt(n, m - 1, 'A'));
            Plan endAtA = simulate(a, b, alt(n - 1, m, 'B'));
            printPlan(endAtB.cost > endAtA.cost ? endAtB : endAtA);
            i128 sumA = 0, sumB = 0;
            for (long long x : a)
                sumA += x;
            for (long long x : b)
                sumB += x;
            Plan mn;
            if (sumA < sumB) {
                mn.cost = sumA;
                for (int i = n - 1; i >= 0; --i)
                    mn.ops.push_back({'A', a[i]});
            } else {
                mn.cost = sumB;
                for (int i = m - 1; i >= 0; --i)
                    mn.ops.push_back({'B', b[i]});
            }
            printPlan(mn);
        }
        return 0;
    }
    ```
