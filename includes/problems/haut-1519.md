    **题目**：第k大分数（简单版）

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1519>){ .problem-source target="_blank" rel="noopener" }

    **题意**：请帮助小EF求出其中第 k 大的分数

    **核心思路**：第k大分数（简单版）暴力基础题 先把所有的形成的分数保存起来，然后按从大到小快排一遍，最后输出第 个即可kkk C++版代码

    **正确性**：推导把原过程转换为等价的公式或不变量。；算法直接计算该等价量，并使用足够宽的数据类型处理边界。；因此计算结果就是题目定义的答案。

    **复杂度**：通常为 O(n log n)，主要来自排序

    **易错点**：比较器必须满足严格弱序；相等时处理题目指定的次关键字。；检查 gcd/lcm 的 0 值、负数和乘法溢出。；检查 0/1 起下标、n = 1、首尾元素和数组容量。；按上界估算中间量，相关变量不要退回 int。

    - 核验状态：C++14 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++14 实现**

    ```cpp linenums="1"
    #include <iostream>
    #include <algorithm>
    #include <vector>
    #define _for(i, a) for (int i = 0; i < (a); ++i)
    #define _rep(i, a, b) for (int i = (a); i <= (b); ++i)
    using namespace std;
    typedef long long LL;
    struct poi {
        LL n, d;
        poi() {}
        poi(int n, int d) : n(n), d(d) {}
        bool operator<(const poi& b) const {
            return 1.0 * n / d > 1.0 * b.n / b.d;
        }
    };
    LL n, k;
    vector<poi> a;
    void sol() {
        _for(i, n) {
            LL tem;
            cin >> tem;
            _rep(i, 1, tem - 1) a.push_back(poi(i, tem));
        }
        sort(a.begin(), a.end());
        cout << a[k - 1].n << "/" << a[k - 1].d << "\n";
    }
    int main() {
        while (cin >> n >> k) {
            sol();
        }
        return 0;
    }
    ```
