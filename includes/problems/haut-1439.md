    **题目**：最大增区间（二）

    [打开 HAUTOJ 原题 ↗](<https://acm.haut.edu.cn/problem.php?id=1439>){ .problem-source target="_blank" rel="noopener" }

    **题意**：求最大增区间的长度。

    **核心思路**：把相邻关系 a&#91;i&#93;≤a&#91;i+1&#93; 记为 1，最长非递减段长度就是最长连续 1 的数量加 1。交换 i、j 只会改变至多四条相邻边；用线段树维护 01 串的最长连续 1，枚举 O(n²) 对交换时只点改这些边、读取根答案， 再回滚。这样保留了题解的“枚举交换”，但避免 O(n³) 重扫。

    **正确性**：枚举范围包含每个可能答案，没有候选被遗漏。；判断条件与题目约束等价，所以只接受合法候选。；按题意计数、取最优或输出这些候选即可得到答案。

    **复杂度**：O(n² log n) 时间、O(n) 空间

    **易错点**：检查 0/1 起下标、n = 1、首尾元素和数组容量。；按上界估算中间量，相关变量不要退回 int。

    - 核验状态：C++14 编译通过；1 个可机读公开样例/合法构造校验通过

    **C++14 实现**

    ```cpp linenums="1"
    #include <bits/stdc++.h>
    using namespace std;
    struct Node {
        int len = 0, prefix = 0, suffix = 0, best = 0;
    };
    Node mergeNode(const Node& left, const Node& right) {
        Node res;
        res.len = left.len + right.len;
        res.prefix = left.prefix == left.len ? left.len + right.prefix : left.prefix;
        res.suffix = right.suffix == right.len ? right.len + left.suffix : right.suffix;
        res.best = max({left.best, right.best, left.suffix + right.prefix});
        return res;
    }
    struct SegmentTree {
        int size = 1;
        vector<Node> tree;
        SegmentTree(const vector<int>& value) {
            while (size < (int)value.size())
                size <<= 1;
            tree.resize(2 * size);
            for (int i = 0; i < size; ++i) {
                int bit = i < (int)value.size() ? value[i] : 0;
                tree[size + i] = {1, bit, bit, bit};
            }
            for (int i = size - 1; i; --i)
                tree[i] = mergeNode(tree[i << 1], tree[i << 1 | 1]);
        }
        void update(int pos, int bit) {
            int node = size + pos;
            tree[node] = {1, bit, bit, bit};
            for (node >>= 1; node; node >>= 1)
                tree[node] = mergeNode(tree[node << 1], tree[node << 1 | 1]);
        }
        int best() const {
            return tree[1].best;
        }
    };
    int main() {
        ios::sync_with_stdio(false);
        cin.tie(nullptr);
        int n;
        cin >> n;
        vector<long long> a(n);
        for (long long& x : a)
            cin >> x;
        if (n == 1) {
            cout << 1 << '\n';
            return 0;
        }
        vector<int> good(n - 1);
        for (int i = 0; i + 1 < n; ++i)
            good[i] = a[i] <= a[i + 1];
        SegmentTree tree(good);
        int ans = tree.best() + 1;
        for (int i = 0; i < n; ++i) {
            for (int j = i + 1; j < n; ++j) {
                vector<int> edges = {i - 1, i, j - 1, j};
                sort(edges.begin(), edges.end());
                edges.erase(unique(edges.begin(), edges.end()), edges.end());
                swap(a[i], a[j]);
                for (int edge : edges) {
                    if (edge >= 0 && edge + 1 < n)
                        tree.update(edge, a[edge] <= a[edge + 1]);
                }
                ans = max(ans, tree.best() + 1);
                swap(a[i], a[j]);
                for (int edge : edges) {
                    if (edge >= 0 && edge + 1 < n)
                        tree.update(edge, good[edge]);
                }
            }
        }
        cout << ans << '\n';
        return 0;
    }
    ```
