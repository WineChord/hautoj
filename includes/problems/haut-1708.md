    **C++14 实现**

    ```cpp linenums="1"
    #include <stdio.h>
    #include <stdlib.h>
    #include <string.h>
    #include <math.h>
    const int MOD = 1e9 + 7;
    int t, n;
    long long F[1000005];
    int main() {
        scanf("%d", &t);
        F[1] = 1, F[2] = 2;
        for (int i = 3; i <= 1000004; i++)
            F[i] = (F[i - 1] % MOD + F[i - 2] % MOD) % MOD;
        while (t--) {
            scanf("%d", &n);
            printf("%lld\n", F[n]);
        }
        return 0;
    }
    ```
